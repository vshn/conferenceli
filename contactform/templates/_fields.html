{% else %} {{ f(class_=classes, **kwargs) }} {% endif %}
{% endmacro %} {% macro select(f) %} {{ field(f, css='input--select', **kwargs) }} {% endmacro %} {% macro textarea(f) %} {{ field(f, css='input--area', **kwargs) }} {% endmacro %} {% macro checkbox(f) %}
